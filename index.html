if ( 'IntersectionObserver' in window && !window.matchMedia('(prefers-reduced-motion: reduce)').matches ) { const items = document.querySelectorAll('.reveal'); const observer = new IntersectionObserver(entries => { entries.forEach(entry => { if (entry.isIntersecting) { entry.target.classList.remove('pending'); observer.unobserve(entry.target); } }); }, { threshold: 0.07 }); document.documentElement.classList.add('motion'); items.forEach(item => { item.classList.add('pending'); observer.observe(item); }); }
